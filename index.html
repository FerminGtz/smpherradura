async function fetchSheetJSON(sheetName) {
// MAGIA NO-STORE PARA DATOS FRESCOS
const response = await fetch(`${baseGviz}&sheet=${sheetName}&t=${new Date().getTime()}`, { cache: 'no-store' });
const text = await response.text();
if (text.includes('"status":"error"')) {
console.warn(`La pestaña "${sheetName}" no se encontró en el Excel.`);
return [];
}
try {
const jsonString = text.substring(text.indexOf("{"), text.lastIndexOf("}") + 1);
const jsonData = JSON.parse(jsonString);
let rows = jsonData.table.rows || [];
let cols = jsonData.table.cols.map(c => (c.label || "").toString().trim());
if ((cols.join("") === "" || cols[0] === "A" || cols[0] === "Col1") && rows.length > 0) {
cols = rows[0].c.map(cell => cell ? (cell.v || cell.f || "").toString().trim() : "");
rows.shift();
}
return rows.map(row => {
let obj = {};
if(row && row.c) {
row.c.forEach((cell, i) => {
let columnName = cols[i] || `Col${i}`;
let val = cell ? (cell.f !== null && cell.f !== undefined ? cell.f : cell.v) : "";
obj[columnName.trim()] = val;
let safeKey = columnName.trim().replace(/\s+/g, '').toLowerCase();
obj[safeKey] = val;
});
}
return obj;
});
} catch (e) {
return [];
}
}

function normalizarNombre(nombre) {
return (nombre || "")
.toString()
.normalize("NFD")
.replace(/[\u0300-\u036f]/g, "")
.replace(/\s+/g, " ")
.trim()
.toLowerCase();
}
function buscarResultadoEntre(jugador1, jugador2) {
const a = normalizarNombre(jugador1);
const b = normalizarNombre(jugador2);
return allMatches.find(m => {
const ganador = normalizarNombre(m.NombreRes);
const perdedor = normalizarNombre(m.Perdedor);
return (ganador === a && perdedor === b) || (ganador === b && perdedor === a);
});
}
function mostrarMisJuegos(programacion) {
const contenedor = document.getElementById("misJuegosContainer");
const intro = document.getElementById("misJuegosIntro");
if (!contenedor || !intro) return;
const jugadorActual = localStorage.getItem("smp_nombre") || "";
if (!jugadorActual) {
intro.innerText = "No fue posible identificar al jugador de la sesión.";
contenedor.innerHTML = "<div class='empty-games'>Vuelve a iniciar sesión para consultar tu programación.</div>";
return;
}
intro.innerText = `Programación de ${jugadorActual}. Puedes adelantar un partido de común acuerdo con tu rival.`;
const actualNorm = normalizarNombre(jugadorActual);
const ordenMes = {
noviembre: 0,
diciembre: 1,
enero: 2,
febrero: 3,
marzo: 4,
abril: 5,
mayo: 6,
junio: 7,
julio: 8,
agosto: 9,
septiembre: 10,
octubre: 11
};
const obtenerOrdenMes = valor => {
const texto = String(valor || "")
.trim()
.toLocaleLowerCase("es")
.normalize("NFD")
.replace(/[\u0300-\u036f]/g, "");
const coincidencia = texto.match(/enero|febrero|marzo|abril|mayo|junio|julio|agosto|septiembre|setiembre|octubre|noviembre|diciembre/);
if (!coincidencia) return 99;
const mes = coincidencia[0] === "setiembre" ? "septiembre" : coincidencia[0];
return ordenMes[mes] ?? 99;
};
const propios = (programacion || [])
.map(p => ({
Mes: p.mes || p.Mes || "",
Orden: Number(p.orden || p.Orden || 999),
Jugador1: p.jugador1 || p.Jugador1 || "",
Jugador2: p.jugador2 || p.Jugador2 || ""
}))
.filter(p => normalizarNombre(p.Jugador1) === actualNorm || normalizarNombre(p.Jugador2) === actualNorm)
.map(p => {
const rival = normalizarNombre(p.Jugador1) === actualNorm ? p.Jugador2 : p.Jugador1;
const resultado = buscarResultadoEntre(jugadorActual, rival);
return {...p, Rival:rival, ResultadoEncontrado:resultado};
})
.sort((a,b) => {
const diferenciaMes = obtenerOrdenMes(a.Mes) - obtenerOrdenMes(b.Mes);
if (diferenciaMes !== 0) return diferenciaMes;
